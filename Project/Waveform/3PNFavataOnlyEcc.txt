(*Modular Eccentric Phase Function*)
PhaseEcc[f_, M_, \[Eta]_, e0_, PNorder_] := 
    Module[{v = (Pi*M*f)^(1/3), 
        v0 = (Pi*M*10)^(1/3),(*Reference frequency 10Hz*)alpha0, 
        leadinge0, leadingList, totalEccPhase}, 
      alpha0 = 3/(128*\[Eta]*v^5);
      leadinge0 = -1/(1462*v^(19/3))*2355*e0^2*v0^(19/3);
      (*Define terms provided in your list*)
      leadingList = {0,(*0.5PN*)
          v0^2 (2833/1008 - (197 \[Eta])/36) + 
            v^2 (299076223/81976608 + (18766963 \[Eta])/
                    2927736),(*1PN*)((377 Pi v0^3)/72 + 
              v^3 (-((2819123 Pi)/
                        282600))),(*1.5PN*)(*... Add your leadinge4 (2PN),
          leadinge5 (2.5PN),leadinge6 (3PN) here...*)};
      (*Slice the list based on PNorder selection*)(*PNorder=
      4 corresponds to 2PN,PNorder=6 corresponds to 3PN*)
      totalEccPhase = 
        alpha0*(leadinge0 + 
              Sum[leadingList[[i]], {i, 1, 
                  Min[PNorder, Length[leadingList]]}]);
      Return[totalEccPhase]];
